#!/usr/bin/env python3
"""Sync chart database admission with the contiguous embedded migration set.

This replaces the removed compatibility-contract.py for the database schema
fields only; agent/protocol compatibility stays unchanged. Run --check in CI.
"""
import argparse
import json
import re
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent


def outputs():
    versions = sorted(int(path.name.split('_', 1)[0]) for path in
                      (ROOT / 'internal/db/migrations').glob('[0-9]*_*.sql'))
    if not versions or versions != list(range(1, len(versions) + 1)):
        raise ValueError('embedded migration versions must be contiguous from 1')
    version = versions[-1]
    result = {}
    for name in ['deploy/chart/values.yaml', 'deploy/testdata/live-values-astronomer-dev.yaml']:
        path = ROOT / name
        text, count = re.subn(r'(?m)^(    targetVersion: )\d+$',
                             lambda match: match[1] + str(version), path.read_text())
        if count != 1:
            raise ValueError(f'{name}: expected exactly one database target')
        text = re.sub(r'drop it or set \d+', f'drop it or set {version}', text)
        result[path] = text
    path = ROOT / 'deploy/chart/values.schema.json'
    text, count = re.subn(r'("targetVersion": \{ "const": )\d+( \})',
                         lambda match: match[1] + str(version) + match[2], path.read_text())
    if count != 2:
        raise ValueError('chart schema: expected ordinary and production target constraints')
    json.loads(text)
    result[path] = text
    path = ROOT / 'deploy/chart/files/release-compatibility.json'
    contract = json.loads(path.read_text())
    contract['postgresql']['target_schema'] = version
    result[path] = json.dumps(contract, indent=2, sort_keys=True) + '\n'
    return result


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--check', action='store_true')
    args = parser.parse_args()
    rendered = outputs()  # Validate every input before writing any output.
    stale = [path for path, text in rendered.items() if path.read_text() != text]
    if args.check and stale:
        parser.exit(1, 'Stale database schema fields: ' + ', '.join(str(p.relative_to(ROOT)) for p in stale) + '\n')
    if not args.check:
        for path in stale:
            path.write_text(rendered[path])
    print('Chart database schema fields match embedded migrations')


if __name__ == '__main__':
    main()
