package main
import("bytes";"encoding/json";"fmt";"go/ast";"go/parser";"go/printer";"go/token";"os")
func symbols(path string)map[string]string{f,e:=parser.ParseFile(token.NewFileSet(),path,nil,parser.SkipObjectResolution);if e!=nil{panic(e)};m:=map[string]string{};ast.Inspect(f,func(n ast.Node)bool{if fld,ok:=n.(*ast.Field);ok{fld.Doc=nil;fld.Comment=nil};return true});print:=func(n ast.Node)string{b:=bytes.Buffer{};printer.Fprint(&b,token.NewFileSet(),n);return b.String()};for _,d:=range f.Decls{switch x:=d.(type){case *ast.GenDecl:for _,s:=range x.Specs{switch t:=s.(type){case *ast.TypeSpec:if ast.IsExported(t.Name.Name){m["type "+t.Name.Name]=print(t.Type)};case *ast.ValueSpec:for _,n:=range t.Names{if ast.IsExported(n.Name){kind:="var ";if x.Tok==token.CONST{kind="const "};if t.Type!=nil{m[kind+n.Name]=print(t.Type)}else{m[kind+n.Name]="inferred"}}}}};case *ast.FuncDecl:if ast.IsExported(x.Name.Name){name:="func "+x.Name.Name;if x.Recv!=nil{name="method "+print(x.Recv.List[0].Type)+"."+x.Name.Name};m[name]=print(x.Type)}}};return m}
func main(){a,b:=symbols(os.Args[1]),symbols(os.Args[2]);removed:=map[string]string{};changed:=map[string][2]string{};added:=0;for k,v:=range a{n,ok:=b[k];if !ok{removed[k]=v}else if n!=v{changed[k]=[2]string{v,n}}};for k:=range b{if _,ok:=a[k];!ok{added++}};r:=map[string]any{"previousExportedDeclarations":len(a),"newExportedDeclarations":len(b),"added":added,"removed":removed,"changed":changed};raw,_:=json.MarshalIndent(r,"","  ");fmt.Println(string(raw))}
