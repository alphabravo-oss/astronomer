"""Draft only: compile, then guarded supplier metadata diagnosis in an owned lane."""
import hashlib
import json
import os
from pathlib import Path
import re
import signal
import stat
import subprocess
import tempfile
import time

ROOT = Path('/root/astronomer-all/astronomer-032-w04')
DRAFT = Path('/tmp/plan032-w04-supplier-metadata-draft/main.go.txt')
sha = lambda data: hashlib.sha256(data).hexdigest()
ENV = os.environ.copy()
ENV.update({'GOTOOLCHAIN': 'go1.27.2', 'GOMAXPROCS': '4', 'GOFLAGS': '-p=2', 'GOWORK': 'off'})


def snapshot():
    commit = subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=ROOT, text=True).strip()
    tree = subprocess.check_output(['git', 'rev-parse', 'HEAD^{tree}'], cwd=ROOT, text=True).strip()
    names = subprocess.check_output(['git', 'ls-files'], cwd=ROOT, text=True).splitlines()
    prefixes = ('internal/delivery/distribution/', 'internal/delivery/resolver/', 'internal/httpclient/', 'internal/jsoncontract/')
    paths = [n for n in names if n in ('go.mod', 'go.sum', 'deploy/bundles/catalog.json', 'scripts/check-build-capacity.sh') or n.startswith(prefixes) and n.endswith('.go') and not n.endswith('_test.go')]
    result = {'commit': commit, 'tree': tree, 'runtimeSourceSHA256': {n: sha((ROOT/n).read_bytes()) for n in paths}}
    for name, digest in result['runtimeSourceSHA256'].items():
        assert sha(subprocess.check_output(['git', 'show', commit+':'+name], cwd=ROOT)) == digest
    return result


def hashed_owned_payload(path, expected):
    before = path.lstat()
    descriptor = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK | os.O_CLOEXEC)
    try:
        opened = os.fstat(descriptor)
        identity = lambda s: (s.st_dev, s.st_ino, s.st_size, s.st_mtime_ns)
        if not stat.S_ISREG(opened.st_mode) or identity(before) != identity(opened) or opened.st_size != expected['Size'] or opened.st_size > 2 << 30:
            raise ValueError('retained payload identity invalid')
        digest = hashlib.sha256()
        count = 0
        while True:
            block = os.read(descriptor, 64 << 10)
            if not block:
                break
            count += len(block)
            if count > expected['Size']:
                raise ValueError('retained payload exceeds claim')
            digest.update(block)
        if count != expected['Size'] or identity(opened) != identity(os.fstat(descriptor)) or identity(opened) != identity(path.lstat()) or 'sha256:'+digest.hexdigest() != expected['Digest']:
            raise ValueError('retained payload hash/identity differs')
        return {'name': path.name, 'size': count, 'sha256': digest.hexdigest()}
    finally:
        os.close(descriptor)


def main():
    assert subprocess.check_output(['git', 'status', '--porcelain'], cwd=ROOT) == b''
    before = snapshot()
    source = DRAFT.read_bytes()
    qualification = Path('/root/.cache/astronomer-032/qualification')
    directory = Path(tempfile.mkdtemp(prefix='w04-supplier-metadata-', dir=qualification))
    directory.chmod(0o700)
    output = directory/'receipt.json'
    receipt = {'schemaVersion': 1, 'kind': 'SupplierMetadataLocalEvidence', 'authoritativeReleaseQualification': False, 'sourceBefore': before, 'runnerSHA256': sha(source), 'driverSHA256': sha(Path(__file__).read_bytes()), 'environment': {k: ENV[k] for k in ('GOTOOLCHAIN', 'GOMAXPROCS', 'GOFLAGS', 'GOWORK')}, 'steps': [], 'retainedPayloads': [], 'retainedMetadata': [], 'productionOIDCPublication': False, 'durableProjection': False, 'measuredScanUsedDB': False}
    def save():
        output.write_text(json.dumps(receipt, indent=2)+'\n'); output.chmod(0o600)
    def capacity_guard(step):
        capacity = directory/(step+'-capacity.log')
        with capacity.open('wb') as log:
            code = subprocess.run(['bash', 'scripts/check-build-capacity.sh'], cwd=ROOT, env=ENV, stdout=log, stderr=subprocess.STDOUT).returncode
        capacity.chmod(0o600)
        receipt['steps'].append({'step': step+'-capacity', 'exitCode': code, 'log': capacity.name, 'sha256': sha(capacity.read_bytes())}); save()
        return code
    def guarded(step, argv, timeout=None):
        code = capacity_guard(step)
        if code:
            return code
        logpath = directory/(step+'.log')
        started = time.time()
        with logpath.open('wb') as log:
            process = subprocess.Popen(argv, cwd=ROOT, env=ENV, stdout=log, stderr=subprocess.DEVNULL if step=='transfer' else subprocess.STDOUT, start_new_session=True)
            try:
                code = process.wait(timeout=timeout)
            except subprocess.TimeoutExpired:
                os.killpg(process.pid, signal.SIGTERM)
                try:
                    code = process.wait(timeout=15)
                except subprocess.TimeoutExpired:
                    os.killpg(process.pid, signal.SIGKILL); code = process.wait()
        logpath.chmod(0o600)
        receipt['steps'].append({'step': step, 'command': argv, 'exitCode': code, 'seconds': round(time.time()-started, 3), 'log': logpath.name, 'sha256': sha(logpath.read_bytes())}); save()
        return code
    # Build closes before network execution; no supplier request is made by build.
    # Preserve exact runner source privately; temporary module source is cleaned.
    (directory/'runner.go.txt').write_bytes(source)
    (directory/'runner.go.txt').chmod(0o600)
    # A fresh immediate child has no pre-existing symlink parent or shared writer.
    temporary_source = Path(tempfile.mkdtemp(prefix='.qualification-supplier-source-', dir=ROOT))
    temporary_source.chmod(0o700)
    try:
        (temporary_source/'main.go').write_bytes(source)
        binary = directory/'supplier-observer'
        code = guarded('build', ['go', 'build', '-o', str(binary), './'+str(temporary_source.relative_to(ROOT))], 300)
    finally:
        (temporary_source/'main.go').unlink(missing_ok=True); temporary_source.rmdir()
    if code:
        receipt['sourceAfter'] = snapshot(); save(); raise SystemExit(1)
    receipt['binarySHA256'] = sha(binary.read_bytes()); save()
    code = guarded('transfer', [str(binary), str(directory)], 4*60)
    raw = directory/'transfer.log'
    try:
        if raw.exists() and raw.stat().st_size:
            observation = json.loads(raw.read_text())
            receipt['observation'] = observation
            for result in observation['results']:
                for exchange in result['exchanges']:
                    name = exchange.get('original_manifest_file')
                    if name:
                        if Path(name).name != name or not re.fullmatch(r'(trivy-database|trivy-java-database)-manifest-[0-9]{1,3}\.json', name):
                            raise ValueError('manifest output name invalid')
                        path = directory/name
                        expected = {'Size': exchange['bytes_read'], 'Digest': 'sha256:'+exchange['original_manifest_sha256']}
                        if expected['Size'] > 1 << 20:
                            raise ValueError('manifest metadata size exceeds bound')
                        receipt['retainedMetadata'].append(hashed_owned_payload(path, expected))
            for result in observation['results']:
                if result['role'] not in ('trivy-database', 'trivy-java-database'):
                    raise ValueError('observation role invalid')
                path = directory/(result['role']+'.tar.gz')
                if path.exists():
                    receipt['retainedPayloads'].append(hashed_owned_payload(path, result['snapshot']['Layer']))
    except (ValueError, KeyError, OSError, TypeError):
        receipt['retainedArtifactBinding'] = 'failed'
        code = 1
    final_capacity = capacity_guard('final')
    if final_capacity:
        code = 1
    receipt['sourceAfter'] = snapshot()
    receipt['sourceBeforeAfterByteEqual'] = receipt['sourceBefore'] == receipt['sourceAfter']
    save()
    # Printed summary contains no token, upstream error or signed URL/query.
    print(json.dumps({'receipt': str(output), 'exitCode': code, 'sourceBeforeAfterByteEqual': receipt['sourceBeforeAfterByteEqual'], 'roles': [{k: result[k] for k in ('role', 'status', 'fresh_at_completion')} for result in receipt.get('observation', {}).get('results', [])]}))
    raise SystemExit(0 if code == 0 and receipt['sourceBeforeAfterByteEqual'] else 1)


if __name__ == '__main__':
    main()
