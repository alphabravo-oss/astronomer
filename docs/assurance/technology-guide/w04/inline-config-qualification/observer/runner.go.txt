const registryOrigin = "https://ghcr.io"
const payloadOrigin = "https://pkg-containers.githubusercontent.com"

// Catalog-selected compatibility is declared intent, not measured scan usage.
var scanner = distribution.DatabaseScannerCompatibility{ImageDigest: "sha256:af6acf9a6b85dfe389a1941505c0ce9efef52a4719635e1a962f022a3d855daa", Version: "0.75.0", MainSchema: 2, JavaSchema: 1}

type exchange struct {
	Method         string `json:"method"`
	Origin         string `json:"origin"`
	Kind           string `json:"kind"`
	Status         int    `json:"status"`
	Range          string `json:"range,omitempty"`
	ContentRange   string `json:"content_range,omitempty"`
	ContentType    string `json:"content_type,omitempty"`
	DeclaredSize   int64  `json:"declared_size"`
	SubjectDigest  string `json:"subject_digest,omitempty"`
	ManifestFile   string `json:"original_manifest_file,omitempty"`
	ManifestSHA256 string `json:"original_manifest_sha256,omitempty"`
	Bytes          int64  `json:"bytes_read"`
}
type observations struct {
	mutex      sync.Mutex
	rows       []*exchange
	base       http.RoundTripper
	root, role string
}

func (observed *observations) RoundTrip(request *http.Request) (*http.Response, error) {
	origin := request.URL.Scheme + "://" + request.URL.Host
	if origin != registryOrigin && origin != payloadOrigin {
		return nil, errors.New("qualification observation origin denied")
	}
	response, err := observed.base.RoundTrip(request)
	row := &exchange{Origin: origin, Kind: "metadata", Method: request.Method, Range: request.Header.Get("Range")}
	if request.URL.Path == "/token" {
		row.Kind = "token"
	} else if row.Range != "" {
		row.Kind = "layer-range"
	}
	if response != nil {
		row.Status = response.StatusCode
		row.DeclaredSize = response.ContentLength
		if mediaClaim.MatchString(response.Header.Get("Content-Type")) {
			row.ContentType = response.Header.Get("Content-Type")
		}
		if digestClaim.MatchString(response.Header.Get("Docker-Content-Digest")) {
			row.SubjectDigest = response.Header.Get("Docker-Content-Digest")
		}
		// Record numeric Content-Range only; no upstream text/header/URL is retained.
		if len(response.Header.Values("Content-Range")) == 1 {
			row.ContentRange = closedRange(response.Header.Get("Content-Range"))
		}
		response.Body = &observedBody{reader: response.Body, owner: observed, row: row, manifest: isOriginalManifest(request) && response.StatusCode == http.StatusOK}
	}
	observed.mutex.Lock()
	observed.rows = append(observed.rows, row)
	observed.mutex.Unlock()
	return response, err
}
func (observed *observations) CloseIdleConnections() {
	if closer, ok := observed.base.(interface{ CloseIdleConnections() }); ok {
		closer.CloseIdleConnections()
	}
}
func (observed *observations) snapshot() []exchange {
	observed.mutex.Lock()
	defer observed.mutex.Unlock()
	rows := make([]exchange, 0, len(observed.rows))
	for _, row := range observed.rows {
		rows = append(rows, *row)
	}
	return rows
}

type observedBody struct {
	reader   io.ReadCloser
	owner    *observations
	row      *exchange
	manifest bool
	content  bytes.Buffer
	closed   bool
}

func (body *observedBody) Read(buffer []byte) (int, error) {
	n, err := body.reader.Read(buffer)
	body.owner.mutex.Lock()
	body.row.Bytes += int64(n)
	if body.manifest && body.content.Len()+n <= 1<<20 {
		_, _ = body.content.Write(buffer[:n])
	} else if body.manifest {
		body.manifest = false
		body.content.Reset()
	}
	body.owner.mutex.Unlock()
	return n, err
}
func (body *observedBody) Close() error {
	err := body.reader.Close()
	body.owner.mutex.Lock()
	defer body.owner.mutex.Unlock()
	if !body.closed && body.manifest && body.content.Len() > 0 && int64(body.content.Len()) == body.row.Bytes {
		name := body.owner.role + "-manifest-" + strconv.Itoa(len(body.owner.rows)) + ".json"
		f, openErr := os.OpenFile(filepath.Join(body.owner.root, name), os.O_CREATE|os.O_EXCL|os.O_WRONLY, 0600)
		if openErr == nil {
			n, writeErr := f.Write(body.content.Bytes())
			closeErr := f.Close()
			if writeErr == nil && closeErr == nil && n == body.content.Len() {
				body.row.ManifestFile = name
				body.row.ManifestSHA256 = fmt.Sprintf("%x", sha256.Sum256(body.content.Bytes()))
			}
		}
	}
	body.closed = true
	return err
}

var digestClaim = regexp.MustCompile(`^sha256:[a-f0-9]{64}$`)
var mediaClaim = regexp.MustCompile(`^[a-z0-9.+-]+/[a-z0-9.+-]{1,192}$`)

func isOriginalManifest(request *http.Request) bool {
	if request.Method != "GET" || request.URL.Scheme != "https" || request.URL.Host != "ghcr.io" || request.URL.RawQuery != "" {
		return false
	}
	for _, repository := range []string{"aquasecurity/trivy-db", "aquasecurity/trivy-java-db"} {
		prefix := "/v2/" + repository + "/manifests/"
		if strings.HasPrefix(request.URL.Path, prefix) && digestClaim.MatchString(strings.TrimPrefix(request.URL.Path, prefix)) {
			return true
		}
	}
	return false
}
func closedRange(value string) string {
	if len(value) > 128 {
		return "invalid"
	}
	for _, character := range value {
		if !strings.ContainsRune("bytes 0123456789-/", character) {
			return "invalid"
		}
	}
	return value
}

type result struct {
	Role              string                                 `json:"role"`
	StartedAt         time.Time                              `json:"started_at"`
	CompletedAt       time.Time                              `json:"completed_at"`
	Status            string                                 `json:"status"`
	Snapshot          *distribution.SupplierDatabaseSnapshot `json:"snapshot,omitempty"`
	MetadataInspected bool                                   `json:"metadata_inspected"`
	FreshAtCompletion bool                                   `json:"fresh_at_completion"`
	Exchanges         []exchange                             `json:"exchanges"`
}
type report struct {
	SchemaVersion             int                                       `json:"schema_version"`
	Kind                      string                                    `json:"kind"`
	StartedAt                 time.Time                                 `json:"started_at"`
	CompletedAt               time.Time                                 `json:"completed_at"`
	Qualification             bool                                      `json:"release_qualification"`
	Scanner                   distribution.DatabaseScannerCompatibility `json:"configured_scanner_compatibility"`
	DNSAddressCounts          map[string]int                            `json:"public_dns_address_counts"`
	Results                   []result                                  `json:"results"`
	ExpandedByteCountMeasured bool                                      `json:"expanded_byte_count_measured"`
	MaximumExpandedBytes      int64                                     `json:"canonical_maximum_expanded_bytes"`
}

func publicSnapshotPolicy(ctx context.Context) (httpclient.DestinationPolicy, map[string]int, error) {
	policy := httpclient.DestinationPolicy{Restricted: true}
	counts := map[string]int{}
	public := httpclient.DestinationPolicy{}
	for _, origin := range []string{registryOrigin, payloadOrigin} {
		if err := public.ValidateURL(ctx, origin); err != nil {
			return policy, counts, errors.New("public origin validation failed")
		}
		host := strings.TrimPrefix(origin, "https://")
		addresses, err := net.DefaultResolver.LookupNetIP(ctx, "ip", host)
		if err != nil || len(addresses) == 0 || len(addresses) > 32 {
			return policy, counts, errors.New("finite public DNS snapshot unavailable")
		}
		trusted := httpclient.TrustedOrigin{URL: origin}
		seen := map[netip.Addr]bool{}
		for _, address := range addresses {
			address = address.Unmap()
			// DNS output cannot grant private/loopback/link-local/metadata exceptions.
			if !public.AllowsAddress(host, address) {
				return policy, counts, errors.New("DNS answer not public-safe")
			}
			if !seen[address] {
				trusted.CIDRs = append(trusted.CIDRs, netip.PrefixFrom(address, address.BitLen()))
				seen[address] = true
			}
		}
		counts[origin] = len(trusted.CIDRs)
		policy.Origins = append(policy.Origins, trusted)
	}
	return policy, counts, policy.Validate()
}

func attempt(ctx context.Context, root, role, repository string, policy httpclient.DestinationPolicy) result {
	outcome := result{Role: role, StartedAt: time.Now().UTC(), Status: "guarded-client-unavailable"}
	client, err := httpclient.NewDestinationClient(policy, httpclient.DestinationOptions{Timeout: time.Minute, MaxResponseBytes: (64 << 20) + 1, MaxRedirects: 2, RedirectOriginPairs: []httpclient.RedirectOriginPair{{From: registryOrigin, To: payloadOrigin}}})
	if err != nil {
		outcome.CompletedAt = time.Now().UTC()
		return outcome
	}
	defer client.CloseIdleConnections()
	observer := &observations{base: client.Transport, root: root, role: role}
	client.Transport = observer
	outcome.Status = "constructor-failed"
	inspector, err := distribution.NewSupplierDatabaseInspector(distribution.SupplierDatabaseOptions{Repository: repository, Role: role, Scanner: scanner, Timing: distribution.CadenceDatabaseUpdateTiming(), Network: policy, Client: client})
	if err == nil {
		outcome.Status = "capture-failed"
		captured, captureError := inspector.Capture(ctx)
		if captureError == nil {
			snapshot, _, _, ok := captured.Snapshot()
			if ok {
				outcome.Snapshot = &snapshot
				outcome.Status = "descriptors-captured-not-payload"
			}
		}
	}
	outcome.CompletedAt = time.Now().UTC()
	outcome.Exchanges = observer.snapshot()
	return outcome
}

func main() {
	if len(os.Args) != 2 {
		os.Exit(2)
	}
	root := os.Args[1]
	info, err := os.Lstat(root)
	if err != nil || !filepath.IsAbs(root) || filepath.Clean(root) != root || !info.IsDir() || info.Mode().Perm() != 0700 {
		os.Exit(2)
	}
	signals, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
	defer stop()
	ctx, cancel := context.WithTimeout(signals, 3*time.Minute)
	defer cancel()
	output := report{SchemaVersion: 1, Kind: "ReviewedSupplierMetadataObservation", StartedAt: time.Now().UTC(), Qualification: false, Scanner: scanner, MaximumExpandedBytes: 8 << 30}
	policy, counts, err := publicSnapshotPolicy(ctx)
	output.DNSAddressCounts = counts
	if err == nil {
		for _, target := range []struct{ role, repository string }{{"trivy-database", "ghcr.io/aquasecurity/trivy-db"}, {"trivy-java-database", "ghcr.io/aquasecurity/trivy-java-db"}} {
			output.Results = append(output.Results, attempt(ctx, root, target.role, target.repository, policy))
		}
	}
	output.CompletedAt = time.Now().UTC()
	if json.NewEncoder(os.Stdout).Encode(output) != nil {
		os.Exit(2)
	}
	if err != nil || len(output.Results) != 2 {
		os.Exit(1)
	}
	for _, outcome := range output.Results {
		if outcome.Status != "descriptors-captured-not-payload" {
			os.Exit(1)
		}
	}
}
