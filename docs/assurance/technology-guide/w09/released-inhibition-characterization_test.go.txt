type w09ReleasedInhibitionReader struct{ AlertingQuerier }

func (w09ReleasedInhibitionReader) GetAlertInhibitionByID(context.Context, uuid.UUID) (sqlc.AlertInhibition, error) {
	return sqlc.AlertInhibition{}, nil
}

func TestW09ReleasedInhibitionCompatibility(t *testing.T) {
	base := `{"name":"synthetic","source_matchers":[{"label":"severity","value":""}],"target_matchers":[{"label":"severity","value":""}],"equal_labels":[]}`
	cases := []struct {
		name, body string
		status     int
		tx         bool
	}{
		{"valid empty value omitted regex", base, 500, true},
		{"missing source", strings.Replace(base, `"source_matchers":[{"label":"severity","value":""}],`, "", 1), 400, false},
		{"empty source", strings.Replace(base, `"source_matchers":[{"label":"severity","value":""}]`, `"source_matchers":[]`, 1), 400, false},
		{"missing target", strings.Replace(base, `"target_matchers":[{"label":"severity","value":""}],`, "", 1), 400, false},
		{"empty target", strings.Replace(base, `"target_matchers":[{"label":"severity","value":""}]`, `"target_matchers":[]`, 1), 400, false},
		{"empty name", strings.Replace(base, `"synthetic"`, `""`, 1), 422, false},
		{"empty label", strings.Replace(base, `"severity"`, `""`, 1), 400, false},
		{"blank label", strings.Replace(base, `"severity"`, `"  "`, 1), 400, false},
		{"missing value", strings.Replace(base, `,"value":""`, "", 1), 500, true},
		{"null value", strings.Replace(base, `"value":""`, `"value":null`, 1), 500, true},
		{"null regex", strings.Replace(base, `"value":""`, `"value":"","is_regex":null`, 1), 500, true},
		{"blank equality label", strings.Replace(base, `"equal_labels":[]`, `"equal_labels":["  "]`, 1), 500, true},
		{"optional nulls", strings.Replace(base, `"equal_labels":[]`, `"equal_labels":null,"enabled":null`, 1), 500, true},
	}
	for _, method := range []string{http.MethodPost, http.MethodPut} {
		for _, tc := range cases {
			t.Run(method+"/"+tc.name, func(t *testing.T) {
				h := NewAlertingHandler(w09ReleasedInhibitionReader{})
				tx := false
				h.SetRunTx(func(context.Context, func(AlertingMutationTx) error) error {
					tx = true
					return errors.New("synthetic transaction boundary")
				})
				r := httptest.NewRequest(method, "/api/v1/admin/alerting/inhibitions/", strings.NewReader(tc.body))
				ctx := chi.NewRouteContext()
				ctx.URLParams.Add("id", "00000000-0000-0000-0000-000000000001")
				r = r.WithContext(context.WithValue(r.Context(), chi.RouteCtxKey, ctx))
				w := httptest.NewRecorder()
				if method == http.MethodPost {
					h.CreateInhibition(w, r)
				} else {
					h.UpdateInhibition(w, r)
				}
				if w.Code != tc.status || tx != tc.tx {
					t.Fatalf("status=%d tx=%t; want status=%d tx=%t", w.Code, tx, tc.status, tc.tx)
				}
			})
		}
	}
}
